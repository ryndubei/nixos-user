  (defun jdt-uri-handler (op &rest args)
    "Handle jdt:// uri's."
    (cond
     ((eq op 'file-remote-p)
      nil)
     ((eq op 'file-readable-p)
      ; TODO: Check with eglot server that the uri actually exist
      t)
     ((eq op 'file-exists-p)
      ; TODO: Check with eglot server that the uri actually exist
      t)
     ((eq op 'file-name-directory)
      "")
     ((eq op 'file-name-nondirectory)
      (car args))
     ((eq op 'expand-file-name)
      (car args))
     ((eq op 'abbreviate-file-name)
      (car args))
     ((eq op 'substitute-in-file-name)
      (car args))
     ((eq op 'file-truename)
      (car args))
     ((eq op 'get-file-buffer)
      (let* ((file (car args))
             (buffer (get-buffer file)))
        (if buffer
            buffer
            (let ((content (jsonrpc-request (eglot--current-server-or-lose) :java/classFileContents `(:uri ,file))))
              (with-current-buffer (create-file-buffer file)
                (insert content)
                (setq buffer-file-name file)
                (setq buffer-read-only t)
                (set-buffer-modified-p nil)
                (goto-char (point-min))
                (java-ts-mode)
                (current-buffer))))))
     (t (jdt-uri-handler-real op args))))

  (defun jdt-uri-handler-real (operation args)
    "Run the real handler without the jdt uri handler installed."
    (let ((inhibit-file-name-handlers
           (cons 'jdt-uri-handler
                 (and (eq inhibit-file-name-operation operation)
                      inhibit-file-name-handlers)))
          (inhibit-file-name-operation operation))
      (apply operation args)))

  (add-to-list 'file-name-handler-alist
               '("\\`jdt://" . jdt-uri-handler)))

#+begin_src emacs-lisp
(use-package emacs-conflict)

(with-eval-after-load 'emacs-conflict
  ;; Monkeypatch emacs-conflict to use the system trash when possible and use
  ;; kill-buffer more safely
  (advice-add 'emacs-conflict--resolve-conflict :override
   (lambda (conflict)
   (let* ((normal (emacs-conflict--get-normal-filename conflict)))
     (emacs-conflict--resolve-ediff
      (list conflict normal)
      (lambda ()
         (when (y-or-n-p "Delete conflict file and save the original file? ")
           (let* ((conflict-buffer (get-file-buffer conflict)))
             (when conflict-buffer
               (kill-buffer conflict-buffer))
             (delete-file conflict t)))))))))

#+begin_src emacs-lisp
;; Every task will have this tag as it will be inherited from filetags, so it should be ignored
(setq org-agenda-hide-tags-regexp "^todo$")

(defun roam-agenda--todo-files ()
  "Return a list of roam files containing the todo filetag"
  (org-roam-db-sync)
  (let ((todo-nodes (seq-filter (lambda (n)
                                  (seq-contains-p (org-roam-node-tags n) "todo"))
                                (org-roam-node-list))))
    (seq-uniq (seq-map #'org-roam-node-file todo-nodes))))

(defun roam-agenda--update-todo-files (&rest _)
  "Set org-agenda-files to all files in org-roam-directory that contain a 'todo' filetag"
  (setq org-agenda-files (roam-agenda--todo-files))
  ;; also add some select non-roam files in org-directory
  (push "Inbox.org" org-agenda-files) ; orgzly
  ;; this file
  (push "~/.config/home-manager/doom.d/config.org" org-agenda-files))

;; Reconstruct org-agenda-files every time an agenda command is dispatched
(with-eval-after-load 'org
  (advice-add 'org-agenda :before #'roam-agenda--update-todo-files))
#+end_src

#+begin_src emacs-lisp
(defun roam-agenda--todo-p ()
  "Return non-nil if the current buffer has a TODO entry"
  (org-element-map
      (org-element-parse-buffer 'headline)
      'headline
    (lambda (h)
      (eq (org-element-property :todo-type h)
          'todo))
    nil 'first-match))

(defun roam-agenda--update-todo-tag ()
  "Update todo tag in the current buffer"
  (when (and
         ;; unsure why checking (not (active-minibuffer-window)) is necessary,
         ;; including this anyway because it's in the article
         (not (active-minibuffer-window))
         (org-roam-file-p))
    ;; add filetags instead of heading tags by moving point to the start of the buffer
    (org-with-point-at 1
      (if (roam-agenda--todo-p)
          (org-roam-tag-add '("todo"))
        (org-roam-tag-remove '("todo"))))))

(define-minor-mode roam-agenda-mode
  "Minor mode that automatically adds 'todo' filetags to org-roam files with TODO headings on save"
  :init-value nil
  (if roam-agenda-mode
      (add-hook 'before-save-hook #'roam-agenda--update-todo-tag nil t)
    (remove-hook 'before-save-hook #'roam-agenda--update-todo-tag t)))

(with-eval-after-load 'org
  ;; note that this will be toggled together with org-mode, i.e.
  ;; org-mode <=> roam-agenda-mode
  (add-hook 'org-mode-hook #'roam-agenda-mode))
#+end_src

* org-roam capture templates

At the end of the config because emacs' elisp parsing for matching parens and
completions dies somewhere here.

Doom Emacs adds some templates to org-capture-templates, but these are not
integrated with org-roam.

(the doom evil binding for capturing to today's daily note is ~SPC n r d T~)

#+begin_src emacs-lisp
(with-eval-after-load 'org
  ;; Default value of org-roam-dailies-capture-templates:
  ;; (("d" "default" entry "* %?" :target
  ;;   (file+head "%<%Y-%m-%d>.org" "#+title: %<%Y-%m-%d>\n")))
  (setq org-roam-dailies-capture-templates
        '(("t" "Personal todo"
           entry
           "* TODO %?\nSCHEDULED: %T"
           :target
           (file+head+olp "%<%Y-%m-%d>.org" "#+STARTUP: overview\n#+title: %<%Y-%m-%d>\n" ("TODOs")))
          ("d" "default"
           entry
           "* %?"
           :target
           (file+head "%<%Y-%m-%d>.org" "#+STARTUP: overview\n#+title: %<%Y-%m-%d>\n"))))
  ;; If you have more than one template, org-roam-dailies-goto-today will
  ;; ask you for the capture template by default. This overrides the
  ;; dailies-goto functions to never prompt for a template interactively
  ;; and to just assume the default (empty) template if one is missing.

  (advice-add #'org-roam-dailies-goto-today :filter-args
              (lambda (args) (apply (lambda (&optional keys) (list (or keys "d"))) args)))
  (advice-add #'org-roam-dailies-goto-tomorrow :filter-args
              (lambda (args) (apply (lambda (n &optional keys) (list n (or keys "d"))) args)))
  (advice-add #'org-roam-dailies-goto-yesterday :filter-args
              (lambda (args) (apply (lambda (n &optional keys) (list n (or keys "d"))) args)))
  (advice-add #'org-roam-dailies-goto-date :filter-args
              (lambda (args) (apply (lambda (&optional prefer-future keys) (list prefer-future (or keys "d"))) args)))

  ;; goto next/previous note don't do the thing and so do not need to be overridden

  ;; TODO change this, this is set to the default value at the moment
  (setq org-roam-capture-templates
        '(("d" "default" plain "%?" :target
          (file+head "%<%Y%m%d%H%M%S>-${slug}.org" "#+title: ${title}\n") :unnarrowed t))))
#+end_src
